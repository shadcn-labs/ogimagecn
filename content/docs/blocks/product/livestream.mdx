## Usage

```tsx
import { ImageResponse } from "next/og";

import { Livestream } from "@/components/og/livestream";
```

```tsx
export function GET() {
  return new ImageResponse(
    <Livestream
      brand="Acme Live"
      title="Building in public, live"
      tagline="Join us for a live session on building better social images."
      platforms={["YouTube", "Twitch"]}
    />,
    { width: 1200, height: 630 }
  );
}
```
