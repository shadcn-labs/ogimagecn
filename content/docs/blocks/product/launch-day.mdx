## Usage

```tsx
import { ImageResponse } from "next/og";

import { LaunchDay } from "@/components/og/launch-day";
```

```tsx
export function GET() {
  return new ImageResponse(
    <LaunchDay
      title="Introducing our new product"
      tagline="The fastest way to ship beautiful, on-brand Open Graph images."
      upvotes="1,204"
      brand="ogimagecn"
    />,
    { width: 1200, height: 630 }
  );
}
```
