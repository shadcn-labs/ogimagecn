## Usage

```tsx
import { ImageResponse } from "next/og";

import { LimitedOffer } from "@/components/og/limited-offer";
```

```tsx
export function GET() {
  return new ImageResponse(
    <LimitedOffer
      brand="ogimagecn"
      code="SAVE50"
      description="Get lifetime access to all components, templates, and future updates at half the price."
      discount="50% OFF"
      endsIn="Ends in 48 hours"
      title="Special Limited Offer"
    />,
    { width: 1200, height: 630 }
  );
}
```
