## Usage

```tsx
import { ImageResponse } from "next/og";

import { Hackathon } from "@/components/og/hackathon";
```

```tsx
export function GET() {
  return new ImageResponse(
    <Hackathon
      dates="Mar 14 – 16, 2026"
      title="Build the future of open source in 48 hours"
      prize="$50,000"
      prizeLabel="in prizes"
      brand="ogimagecn"
    />,
    { width: 1200, height: 630 }
  );
}
```
