## Usage

```tsx
import { ImageResponse } from "next/og";

import { CommunityAma } from "@/components/og/community-ama";
```

```tsx
export function GET() {
  return new ImageResponse(
    <CommunityAma
      label="Community AMA"
      title="Ask me anything about open source"
      guest="Ada Lovelace"
      role="Maintainer of ogimagecn"
      date="Thu, Jul 9 · 5:00 PM UTC"
      brand="ogimagecn"
    />,
    { width: 1200, height: 630 }
  );
}
```
