## Usage

```tsx
import { ImageResponse } from "next/og";

import { MinimalBrand } from "@/components/og/minimal-brand";
```

```tsx
export function GET() {
  return new ImageResponse(
    <MinimalBrand
      brand="ogimagecn"
      tagline="Open Graph images, built on Satori"
    />,
    { width: 1200, height: 630 }
  );
}
```
