## Usage

```tsx
import { ImageResponse } from "next/og";

import { VideoSeries } from "@/components/og/video-series";
```

```tsx
export function GET() {
  return new ImageResponse(
    <VideoSeries
      title="Building Open Graph images"
      episodes={[
        "Getting started with Satori",
        "Composing a card from registry components",
        "Shipping to the edge",
      ]}
      episodeCount="12 episodes"
      brand="ogimagecn"
    />,
    { width: 1200, height: 630 }
  );
}
```
