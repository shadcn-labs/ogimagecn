## Usage

```tsx
import { ImageResponse } from "next/og";

import { Recipe } from "@/components/og/recipe";
```

```tsx
export function GET() {
  return new ImageResponse(
    <Recipe
      title="One-pot tomato basil pasta"
      description="Blistered tomatoes, garlic, and basil tossed with bucatini in one pan."
      duration="25 min"
      servings="Serves 4"
      brand="ogimagecn"
    />,
    { width: 1200, height: 630 }
  );
}
```
